cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(ultraterimal VERSION 0.1.0 LANGUAGES CXX)
option(ULTRATERIMAL_BUILD_AGENT "Build the Windows Agent" ${WIN32})
if(ULTRATERIMAL_BUILD_AGENT AND (NOT WIN32 OR NOT MSVC))
    message(FATAL_ERROR "The Agent requires Windows and MSVC; disable ULTRATERIMAL_BUILD_AGENT for server-only builds.")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
option(ULTRATERIMAL_BUILD_SERVER "Build the Ruvia gateway and MariaDB platform" OFF)
include(FetchContent)
set(RUVIA_BUILD_CORE ON CACHE BOOL "" FORCE)
set(RUVIA_BUILD_HTTP ${ULTRATERIMAL_BUILD_SERVER} CACHE BOOL "" FORCE)
set(RUVIA_BUILD_WEB ${ULTRATERIMAL_BUILD_SERVER} CACHE BOOL "" FORCE)
set(RUVIA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RUVIA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RUVIA_ENABLE_MARIADB OFF CACHE BOOL "" FORCE)
set(RUVIA_ENABLE_POSTGRESQL OFF CACHE BOOL "" FORCE)
set(RUVIA_ENABLE_REDIS OFF CACHE BOOL "" FORCE)
set(RUVIA_ENABLE_JWT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ruvia
    SYSTEM
    GIT_REPOSITORY https://github.com/hyird/Ruvia.git
    GIT_TAG 50e408310df15a1c7d9e7c6e241715f49a0c6548)
FetchContent_MakeAvailable(ruvia)
add_subdirectory(protocol)
if(ULTRATERIMAL_BUILD_AGENT)
    add_subdirectory(agent)
endif()
if(ULTRATERIMAL_BUILD_SERVER)
    add_subdirectory(server)
endif()
if(BUILD_TESTING)
    if(ULTRATERIMAL_BUILD_AGENT)
        add_subdirectory(tests/agent)
        add_subdirectory(tests/web)
    endif()
    add_subdirectory(tests/protocol)
    if(ULTRATERIMAL_BUILD_SERVER)
        add_subdirectory(tests/server)
    endif()
endif()
if(ULTRATERIMAL_BUILD_AGENT)
    install(TARGETS ultraterimal-agent RUNTIME DESTINATION .)
    install(FILES agent/platform/windows/install_agent.ps1 DESTINATION .)
endif()
