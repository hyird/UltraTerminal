cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(ultraterimal VERSION 0.1.0 LANGUAGES CXX)
option(ULTRATERIMAL_BUILD_AGENT "Build the Windows Agent" ${WIN32})
if(ULTRATERIMAL_BUILD_AGENT AND (NOT WIN32 OR NOT MSVC))
    message(FATAL_ERROR "The Agent requires Windows and MSVC; disable ULTRATERIMAL_BUILD_AGENT for server-only builds.")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
option(ULTRATERIMAL_BUILD_SERVER "Build the Ruvia gateway and PostgreSQL platform" OFF)
add_subdirectory(protocol)
if(ULTRATERIMAL_BUILD_AGENT)
    add_subdirectory(agent)
endif()
if(ULTRATERIMAL_BUILD_SERVER)
    add_subdirectory(server)
endif()
if(BUILD_TESTING)
    if(ULTRATERIMAL_BUILD_AGENT)
        add_subdirectory(tests/agent)
        add_subdirectory(tests/web)
    endif()
    add_subdirectory(tests/protocol)
    if(ULTRATERIMAL_BUILD_SERVER)
        add_subdirectory(tests/server)
    endif()
endif()
if(ULTRATERIMAL_BUILD_AGENT)
    install(TARGETS ultraterimal-agent RUNTIME DESTINATION .)
    install(FILES agent/platform/windows/install_agent.ps1 DESTINATION .)
endif()
